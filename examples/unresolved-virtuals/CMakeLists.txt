cmake_minimum_required(VERSION 3.20)
project(unresolved_virtuals LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(virtual_consumer SHARED src/consumer.cpp)
target_include_directories(virtual_consumer PRIVATE include)
target_compile_options(virtual_consumer PRIVATE -fno-inline)
if(APPLE)
  target_link_options(virtual_consumer PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()
add_executable(loader_probe src/loader_probe.cpp)
target_link_libraries(loader_probe PRIVATE ${CMAKE_DL_LIBS})
