cmake_minimum_required(VERSION 3.20)
project(unresolved_callables LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(callable_consumer SHARED src/consumer.cpp)
target_include_directories(callable_consumer PRIVATE include)
if(APPLE)
  target_link_options(callable_consumer PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()
add_executable(loader_probe src/loader_probe.cpp)
target_link_libraries(loader_probe PRIVATE ${CMAKE_DL_LIBS})
